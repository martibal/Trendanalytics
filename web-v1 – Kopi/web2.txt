1.1 Forbidden Language Detection System [LEGAL]
Requirement: Implement automated content scanning system
Implementation:
javascript// File: /src/utils/legalCompliance.js

const FORBIDDEN_TERMS = {
  predictive: [
    'will', 'should', 'expect to', 'likely to', 'predicted',
    'forecasted', 'anticipated', 'projected', 'going to',
    'about to', 'set to', 'poised to', 'destined to'
  ],
  advisory: [
    'you should', 'we recommend', 'consider buying',
    'consider selling', 'good time to', 'opportunity',
    'favorable', 'buy now', 'sell now', 'hold',
    'enter position', 'exit position'
  ],
  sentiment: [
    'bullish', 'bearish', 'moon', 'dump', 'pump',
    'rekt', 'fud', 'fomo', 'hopium'
  ],
  causal_price: [
    'will affect price', 'indicates price movement',
    'signals market direction', 'suggests trend in value',
    'price will', 'value will', 'market will'
  ]
};

const APPROVED_PATTERNS = {
  descriptive: [
    'is', 'was', 'has been', 'observed', 'measured',
    'recorded', 'currently shows', 'historically ranked'
  ],
  comparative: [
    'higher than', 'lower than', 'compared to',
    'relative to', 'in the Xth percentile'
  ],
  conditional: [
    'if conditions persist', 'in similar historical periods',
    'when this has occurred before', 'historically'
  ]
};

export function scanForForbiddenLanguage(text) {
  const violations = [];
  const lowerText = text.toLowerCase();
  
  Object.entries(FORBIDDEN_TERMS).forEach(([category, terms]) => {
    terms.forEach(term => {
      if (lowerText.includes(term)) {
        violations.push({
          category,
          term,
          severity: 'CRITICAL',
          message: `Forbidden ${category} language detected: "${term}"`
        });
      }
    });
  });
  
  return {
    isCompliant: violations.length === 0,
    violations
  };
}

// Pre-commit hook integration
export function validateContentCompliance(content) {
  const result = scanForForbiddenLanguage(content);
  
  if (!result.isCompliant) {
    console.error('LEGAL COMPLIANCE VIOLATION:');
    result.violations.forEach(v => {
      console.error(`  - ${v.message}`);
    });
    throw new Error('Content failed legal compliance check');
  }
  
  return true;
}

javascriptexport function addLegalDisclaimers(req, res, next) {
  const originalJson = res.json.bind(res);
  
  res.json = function(data) {
    const wrappedData = {
      ...data,
      disclaimer: "Descriptive data only. Does not constitute financial advice or prediction of future conditions.",
      timestamp: new Date().toISOString(),
      legal_notice: "By using this API, you agree to the Terms of Service at https://css.example.com/terms"
    };
    
    return originalJson(wrappedData);
  };
  
  next();
}

2.2 Chain Profile Loader Service
File: /src/services/chainProfileService.js
javascriptimport chainProfiles from '@/config/chainProfiles.json';

class ChainProfileService {
  constructor() {
    this.profiles = chainProfiles;
  }
  
  /**
   * Get complete profile for a chain
   * @param {string} chainId - Chain identifier (ethereum, bitcoin, arbitrum, base)
   * @returns {Object} Complete chain profile
   */
  getProfile(chainId) {
    const profile = this.profiles[chainId];
    if (!profile) {
      throw new Error(`Chain profile not found: ${chainId}`);
    }
    return profile;
  }
  
  /**
   * Get ordered primary metrics for a chain
   * @param {string} chainId
   * @returns {Array} Ordered array of primary metrics
   */
  getPrimaryMetrics(chainId) {
    const profile = this.getProfile(chainId);
    return profile.metrics.primary.sort((a, b) => a.order - b.order);
  }
  
  /**
   * Get ordered secondary metrics for a chain
   * @param {string} chainId
   * @returns {Array} Ordered array of secondary metrics
   */
  getSecondaryMetrics(chainId) {
    const profile = this.getProfile(chainId);
    return profile.metrics.secondary.sort((a, b) => a.order - b.order);
  }
  
  /**
   * Check if a metric should be displayed for a chain
   * @param {string} chainId
   * @param {string} metricKey
   * @returns {boolean}
   */
  isMetricRelevant(chainId, metricKey) {
    const profile = this.getProfile(chainId);
    
    // Check if in hidden list
    if (profile.metrics.hidden.includes(metricKey)) {
      return false;
    }
    
    // Check if in primary or secondary
    const allMetrics = [
      ...profile.metrics.primary,
      ...profile.metrics.secondary
    ];
    
    return allMetrics.some(m => m.key === metricKey);
  }
  
  /**
   * Get metric definition for a specific chain
   * @param {string} chainId
   * @param {string} metricKey
   * @returns {Object|null} Metric definition or null if not found
   */
  getMetricDefinition(chainId, metricKey) {
    const profile = this.getProfile(chainId);
    
    const allMetrics = [
      ...profile.metrics.primary,
      ...profile.metrics.secondary
    ];
    
    return allMetrics.find(m => m.key === metricKey) || null;
  }
  
  /**
   * Get interpretation context for a chain
   * @param {string} chainId
   * @returns {Object} Interpretation object with lens, focus, context

3.4 API Endpoint for Recent Regime Shift
File: /src/api/routes/landingData.js
javascriptimport express from 'express';
import { addLegalDisclaimers } from '../middleware/legalMiddleware';
import chainProfileService from '@/services/chainProfileService';

const router = express.Router();
router.use(addLegalDisclaimers);

/**
 * GET /api/landing/recent-regime-shift
 * Returns data for the most recent regime shift across all chains
 */
router.get('/recent-regime-shift', async (req, res) => {
  try {
    // Query database for most recent regime shift
    const recentShift = await db.query(`
      SELECT 
        chain,
        shift_date,
        previous_regime,
        current_regime,
        metric_key
      FROM regime_shifts
      ORDER BY shift_date DESC
      LIMIT 1
    `);
    
    if (!recentShift) {
      return res.status(404).json({ error: 'No recent regime shifts found' });
    }
    
    // Get 30 days of data before and after shift
    const startDate = addDays(recentShift.shift_date, -30);
    const endDate = addDays(recentShift.shift_date, 30);
    
    const timeSeriesData = await db.query(`
      SELECT 
        date,
        ${recentShift.metric_key} as value,
        ${recentShift.metric_key}__ma7 as ma7,
        ${recentShift.metric_key}__ma30 as ma30
      FROM chain_data
      WHERE chain = ? AND date BETWEEN ? AND ?
      ORDER BY date ASC
    `, [recentShift.chain, startDate, endDate]);
    
    // Get chain profile for color
    const profile = chainProfileService.getProfile(recentShift.chain);
    
    // Format response
    const response = {
      chain_id: recentShift.chain,
      chain_name: profile.name,
      chain_color: profile.color,
      shift_date: recentShift.shift_date,
      previous_regime: recentShift.previous_regime,
      current_regime: recentShift.current_regime,
      metric: recentShift.metric_key,
      dates: timeSeriesData.map(d => d.date),
      ma7_values: timeSeriesData.map(d => d.ma7),
      ma30_values: timeSeriesData.map(d => d.ma30),
      shift_date_index: timeSeriesData.findIndex(d => d.date === recentShift.shift_date),
      direction: recentShift.current_regime === 'CONGESTED' ? 'up' : 'down'
    };
    
    res.json(response);
    
  } catch (error) {
    console.error('Error fetching recent regime shift:', error);
    res.status(500).json({ error: 'Internal server error' });
  }
});

function addDays(date, days) {
  const result = new Date(date);
  result.setDate(result.getDate() + days);
  return result.toISOString().split('T')[0];
}

export default router;

function getPercentileLabel(percentile) {
  if (percentile >= 90) return 'Extreme';
  if (percentile >= 75) return 'Very High';
  if (percentile >= 60) return 'High';
  if (percentile >= 40) return 'Normal';
  if (percentile >= 25) return 'Low';
  if (percentile >= 10) return 'Very Low';
  return 'Extreme Low';
}

function getTrendText(current, ma7, ma30) {
  const vsMA7 = ((current - ma7) / ma7 * 100).toFixed(1);
  const direction = vsMA7 >= 0 ? '+' : '';
  return `${direction}${vsMA7}% vs MA7`;
}

  // This would be more sophisticated in production,

  return `${metric.label} is ${level.toLowerCase()} relative to the past 90 days, ranking higher than ${percentile}% of observed days.`;
}

javascript/**
 * Format metric value based on type
 */
export function formatMetricValue(value, format) {
  if (value === null || value === undefined) {
    return 'N/A';
  }
  
  switch (format) {
    case 'number':
      return formatNumber(value);
    
    case 'percentage':
      return formatPercentage(value);
    
    case 'gwei':
      return formatGwei(value);
    
    case 'eth':
      return formatEth(value);
    
    case 'btc':
      return formatBtc(value);
    
    case 'satoshis':
      return formatSatoshis(value);
    
    case 'seconds':
      return formatSeconds(value);
    
    case 'ratio':
      return formatRatio(value);
    
    default:
      return formatNumber(value);
  }
}

/**
 * Format large numbers with K/M/B suffixes
 */
export function formatNumber(num) {
  if (num >= 1_000_000_000) {
    return (num / 1_000_000_000).toFixed(2) + 'B';
  }
  if (num >= 1_000_000) {
    return (num / 1_000_000).toFixed(2) + 'M';
  }
  if (num >= 1_000) {
    return (num / 1_000).toFixed(2) + 'K';
  }
  return num.toFixed(0);
}

/**
 * Format percentage values
 */
export function formatPercentage(value) {
  return (value * 100).toFixed(2) + '%';
}

/**
 * Format Gwei values
 */
export function formatGwei(value) {
  const gwei = value / 1_000_000_000;
  if (gwei >= 1000) {
    return gwei.toFixed(0) + ' Gwei';
  }
  return gwei.toFixed(2) + ' Gwei';
}

/**
 * Format ETH values
 */
export function formatEth(value) {
  const eth = value / 1e18;
  if (eth >= 1_000_000) {
    return (eth / 1_000_000).toFixed(2) + 'M ETH';
  }
  if (eth >= 1_000) {
    return (eth / 1_000).toFixed(2) + 'K ETH';
  }
  return eth.toFixed(4) + ' ETH';
}

/**
 * Format BTC values
 */
export function formatBtc(value) {
  const btc = value / 1e8;  // satoshis to BTC
  if (btc >= 1_000_000) {
    return (btc / 1_000_000).toFixed(2) + 'M BTC';
  }
  if (btc >= 1_000) {
    return (btc / 1_000).toFixed(2) + 'K BTC';
  }
  return btc.toFixed(4) + ' BTC';
}

/**
 * Format satoshis
 */
export function formatSatoshis(value) {
  if (value >= 100_000_000) {
    return (value / 100_000_000).toFixed(2) + ' BTC';
  }
  return formatNumber(value) + ' sats';
}

/**
 * Format seconds
 */
export function formatSeconds(value) {
  if (value >= 60) {
    const minutes = Math.floor(value / 60);
    const seconds = Math.floor(value % 60);
    return `${minutes}m ${seconds}s`;
  }
  return value.toFixed(1) + 's';
}

/**
 * Format ratio values
 */
export function formatRatio(value) {
  return value.toFixed(2);
}

/**
 * Format values for chart display
 */
export function formatChartValue(value, metricKey) {
  // Simplified formatting for chart tooltips and axes
  if (metricKey.includes('fee')) {
    return formatGwei(value);
  }
  if (metricKey.includes('value_transferred')) {
    if (metricKey.includes('eth')) {
      return formatEth(value);
    }
    if (metricKey.includes('btc')) {
      return formatBtc(value);
    }
  }
  if (metricKey.includes('utilization') || metricKey.includes('rate')) {
    return formatPercentage(value);
  }
  if (metricKey.includes('block_time')) {
    return formatSeconds(value);
  }
  return formatNumber(value);
}

/**
 * Get trend icon based on direction
 */
export function getTrendIcon(trend) {
  if (trend === 'HEATING' || trend > 0) return '↑';
  if (trend === 'COOLING' || trend < 0) return '↓';
  return '→';
}

/**
 * Get trend color based on direction
 */
export function getTrendColor(trend) {
  if (trend === 'HEATING' || trend > 5) return '#FF4A5F';
  if (trend === 'COOLING' || trend < -5) return '#4AFF8C';
  return 'rgba(228, 231, 235, 0.6)';
}

/**
 * Get regime background color
 */
export function getRegimeColor(regime, opacity = 1.0) {
  const colors = {
    'CONGESTED': `rgba(255, 74, 95, ${opacity})`,
    'NORMAL': `rgba(255, 184, 74, ${opacity})`,
    'COOLING': `rgba(74, 255, 140, ${opacity})`
  };
  return colors[regime] || `rgba(228, 231,Continue10:56 PM235, ${opacity})`;
}

## 6. NARRATIVE GENERATION SYSTEM

### 6.1 Narrative Generator Service

**File:** `/src/services/narrativeGenerator.js`
```javascript
import chainProfileService from './chainProfileService';
import { formatMetricValue } from '@/utils/formatters';

class NarrativeGenerator {
  
  /**
   * Generate complete narrative for a chain
   * @param {string} chainId
   * @param {Object} data - Complete chain data including regime, metrics, scorecard
   * @returns {string} Generated narrative
   */
  generate(chainId, data) {
    const profile = chainProfileService.getProfile(chainId);
    const template = profile.narrative_template;
    
    // Build narrative sections
    const sections = [
      this._generateOpening(chainId, data, template),
      this._generateDemandSection(chainId, data, template),
      this._generateCapacitySection(chainId, data, template),
      this._generateFrictionSection(chainId, data, template),
      this._generateHistorySection(chainId, data, template)
    ];
    
    // Join with paragraph breaks
    return sections.filter(s => s).join('\n\n');
  }
  
  /**
   * Generate opening statement
   */
  _generateOpening(chainId, data, template) {
    const chainName = chainProfileService.getProfile(chainId).name;
    const regime = data.regime.label;
    
    return template.opening
      .replace('{chain}', chainName)
      .replace('{regime}', regime);
  }
  
  /**
   * Generate demand section (chain-specific)
   */
  _generateDemandSection(chainId, data, template) {
    const profile = chainProfileService.getProfile(chainId);
    const primaryMetric = profile.metrics.primary[0];
    
    const value = data[primaryMetric.key];
    const formattedValue = formatMetricValue(value, primaryMetric.format);
    const percentile = this._getPercentile(data, primaryMetric.key);
    const trend = this._getTrendDirection(data, primaryMetric.key);
    
    const interpretation = this._getInterpretation(value, percentile);
    
    let section = template.demand_section
      .replace('{demand_interpretation}', interpretation)
      .replace('{key_metric}', primaryMetric.label.toLowerCase())
      .replace('{value}', formattedValue)
      .replace('{percentile}', percentile)
      .replace('{trend_direction}', trend);
    
    // Chain-specific additions
    if (chainId === 'base') {
      const growthRate = this._calculateGrowthRate(data, primaryMetric.key);
      section = section.replace('{wow_growth}', growthRate.toFixed(1));
    }
    
    return section;
  }
  
  /**
   * Generate capacity section (chain-specific)
   */
  _generateCapacitySection(chainId, data, template) {
    const profile = chainProfileService.getProfile(chainId);
    
    // Different capacity metrics for different chains
    let section = template.capacity_section;
    
    if (chainId === 'ethereum') {
      const gasUtil = data.gas_utilization_pct;
      const utilInterpretation = this._getUtilizationInterpretation(gasUtil);
      const baseline = this._compareToBaseline(
        data.gas_utilization_pct,
        data.derived.metrics.gas_utilization_pct__ma30
      );
      
      section = section
        .replace('{utilization_interpretation}', utilInterpretation)
        .replace('{gas_util}', (gasUtil * 100).toFixed(1) + '%')
        .replace('{baseline_comparison}', baseline);
        
    } else if (chainId === 'bitcoin') {
      const blockTime = data.avg_block_time_sec;
      const blockTimeInterpretation = this._getBlockTimeInterpretation(blockTime);
      
      section = section
        .replace('{block_time_interpretation}', blockTimeInterpretation)
        .replace('{block_time}', blockTime.toFixed(1) + ' seconds');
        
    } else if (chainId === 'arbitrum' || chainId === 'base') {
      const failedRate = data.failed_tx_rate;
      const failedInterpretation = this._getFailedTxInterpretation(failedRate);
      
      section = section
        .replace('{failed_tx_rate}', (failedRate * 100).toFixed(2))
        .replace('{failed_tx_interpretation}', failedInterpretation);
    }
    
    return section;
  }
  
  /**
   * Generate friction section
   */
  _generateFrictionSection(chainId, data, template) {
    const feeValue = data.median_tx_fee_native;
    const feeFormat = chainId === 'bitcoin' ? 'satoshis' : 'gwei';
    const formattedFee = formatMetricValue(feeValue, feeFormat);
    
    const feeTrend = this._getTrendDirection(data, 'median_tx_fee_native');
    const feeInterpretation = this._getFeeInterpretation(
      data.median_tx_fee_native,
      data.derived.metrics.median_tx_fee_native__ma30
    );
    
    const failedTxRate = data.failed_tx_rate;
    const failedInterpretation = this._getFailedTxInterpretation(failedTxRate);
    
    return template.friction_section
      .replace('{fee_interpretation}', feeInterpretation)
      .replace('{fee_value}', formattedFee)
      .replace('{fee_trend}', feeTrend)
      .replace('{failed_tx_interpretation}', failedInterpretation);
  }
  
  /**
   * Generate history section
   */
  _generateHistorySection(chainId, data, template) {
    const chainName = chainProfileService.getProfile(chainId).name;
    const duration = data.regime_duration_days || 1;
    const lastShiftDate = data.last_regime_shift_date || 'N/A';
    const previousRegime = data.previous_regime || 'UNKNOWN';
    const currentRegime = data.regime.label;
    
    return template.history_section
      .replace('{duration}', duration)
      .replace('{last_shift_date}', this._formatDate(lastShiftDate))
      .replace('{chain}', chainName)
      .replace('{previous_regime}', previousRegime)
      .replace('{current_regime}', currentRegime);
  }
  
  /**
   * Helper: Get percentile for a metric
   */
  _getPercentile(data, metricKey) {
    if (data.derived && data.derived.percentiles) {
      return data.derived.percentiles[metricKey] || 50;
    }
    return 50;
  }
  
  /**
   * Helper: Get trend direction
   */
  _getTrendDirection(data, metricKey) {
    const current = data[metricKey];
    const ma7 = data.derived.metrics[`${metricKey}__ma7`];
    
    if (!ma7 || current === ma7) return 'stable';
    
    const change = ((current - ma7) / ma7) * 100;
    
    if (change > 5) return 'heating';
    if (change < -5) return 'cooling';
    return 'stable';
  }
  
  /**
   * Helper: Get interpretation based on value and percentile
   */
  _getInterpretation(value, percentile) {
    if (percentile >= 90) return 'extremely elevated';
    if (percentile >= 75) return 'significantly elevated';
    if (percentile >= 60) return 'elevated';
    if (percentile >= 40) return 'typical';
    if (percentile >= 25) return 'below normal';
    if (percentile >= 10) return 'significantly below normal';
    return 'extremely low';
  }
  
  /**
   * Helper: Get utilization interpretation
   */
  _getUtilizationInterpretation(utilization) {
    if (utilization >= 0.9) return 'operating at near-maximum capacity';
    if (utilization >= 0.7) return 'operating under elevated capacity pressure';
    if (utilization >= 0.5) return 'operating at balanced capacity';
    if (utilization >= 0.3) return 'operating with significant capacity headroom';
    return 'operating well below capacity';
  }
  
  /**
   * Helper: Compare to baseline
   */
  _compareToBaseline(current, baseline) {
    if (!baseline) return 'No baseline available';
    
    const diff = ((current - baseline) / baseline) * 100;
    
    if (Math.abs(diff) < 5) return 'roughly in line with the 30-day baseline';
    if (diff > 0) return `${diff.toFixed(1)}% above the 30-day baseline`;
    return `${Math.abs(diff).toFixed(1)}% below the 30-day baseline`;
  }
  
  /**
   * Helper: Get block time interpretation
   */
  _getBlockTimeInterpretation(blockTime) {
    const target = 600;  // 10 minutes in seconds
    const diff = ((blockTime - target) / target) * 100;
    
    if (Math.abs(diff) < 5) return 'operating at target pace';
    if (diff < 0) return `faster than target (${Math.abs(diff).toFixed(1)}% faster)`;
    return `slower than target (${diff.toFixed(1)}% slower)`;
  }
  
  /**
   * Helper: Get failed tx interpretation
   */
  _getFailedTxInterpretation(rate) {
    if (rate >= 0.05) return 'elevated, indicating congestion stress';
    if (rate >= 0.02) return 'slightly elevated but manageable';
    if (rate >= 0.01) return 'normal and expected';
    return 'very low, indicating smooth operations';
  }
  
  /**
   * Helper: Get fee interpretation
   */
  _getFeeInterpretation(current, baseline) {
    if (!baseline) return 'typical for current conditions';
    
    const diff = ((current - baseline) / baseline) * 100;
    
    if (diff > 50) return 'significantly elevated';
    if (diff > 20) return 'moderately elevated';
    if (diff < -20) return 'below recent norms';
    return 'stable and typical';
  }
  
  /**
   * Helper: Calculate growth rate
   */
  _calculateGrowthRate(data, metricKey) {
    const current = data.derived.metrics[`${metricKey}__ma7`];
    const previous = data.derived.metrics[`${metricKey}__ma30`];
    
    if (!current || !previous) return 0;
    
    return ((current - previous) / previous) * 100;
  }
  
  /**
   * Helper: Format date
   */
  _formatDate(dateString) {
    if (!dateString || dateString === 'N/A') return 'N/A';
    
    const date = new Date(dateString);
    return date.toLocaleDateString('en-US', {
      month: 'long',
      day: 'numeric',
      year: 'numeric'
    });
  }
}

// Export singleton
export default new NarrativeGenerator();
```

```jsx
import React from 'react';

export default function Button({
  children,
  variant = 'primary',  // 'primary' | 'secondary' | 'ghost'
  size = 'medium',      // 'small' | 'medium' | 'large'
  fullWidth = false,
  disabled = false,
  onClick,

  const disabledClass = disabled ? 'btn-disabled' : '';

    baseClass,
    variantClass,
    sizeClass,
    widthClass,
    disabledClass,
    className
  ].filter(Boolean).join(' ');
  
  return (
    <button
      type={type}
      className={classes}
      onClick={onClick}
      disabled={disabled}
    >
      {icon && iconPosition === 'left' && (
        <span className="btn-icon btn-icon-left">{icon}</span>
      )}
      <span className="btn-text">{children}</span>
      {icon && iconPosition === 'right' && (
        <span className="btn-icon btn-icon-right">{icon}</span>
      )}
    </button>
  );
}
```

```javascript
import express from 'express';
import db from '@/db';
import chainProfileService from '@/services/chainProfileService';

const router = express.Router();

/**
 * GET /api/chains/:chainId/latest
 * Returns the most recent data for a chain
 */
router.get('/:chainId/latest', async (req, res) => {
  try {
    const { chainId } = req.params;
    
    // Validate chain ID
    const validChains = ['ethereum', 'bitcoin', 'arbitrum', 'base'];
    if (!validChains.includes(chainId)) {
      return res.status(400).json({
        error: `Invalid chain ID. Must be one of: ${validChains.join(', ')}`
      });
    }
    
    // Fetch latest data
    const latestData = await db.getLatestChainData(chainId);
    
    if (!latestData) {
      return res.status(404).json({
        error: `No data found for chain: ${chainId}`
      });
    }
    
    // Get chain profile for context
    const profile = chainProfileService.getProfile(chainId);
    
    // Format response
    const response = {
      chain: chainId,
      chain_name: profile.name,
      as_of_date: latestData.date,
      regime: latestData.regime,
      scorecard: latestData.scorecard,
      metrics: latestData,
      derived: latestData.derived,
      interpretation_context: profile.interpretation
    };
    
    res.json(response);
    
  } catch (error) {
    console.error('Error fetching chain data:', error);
    res.status(500).json({
      error: 'Failed to fetch chain data'
    });
  }
});

/**
 * GET /api/chains/:chainId/historical
 * Returns historical data for a chain
 * Query params:
 *   - start_date: YYYY-MM-DD
 *   - end_date: YYYY-MM-DD
 *   - window: 7d, 30d, 90d, 180d, 365d, all
 */
router.get('/:chainId/historical', async (req, res) => {
  try {
    const { chainId } = req.params;
    const { start_date, end_date, window = '90d' } = req.query;
    
    // Validate chain ID
    const validChains = ['ethereum', 'bitcoin', 'arbitrum', 'base'];
    if (!validChains.includes(chainId)) {
      return res.status(400).json({
        error: `Invalid chain ID. Must be one of: ${validChains.join(', ')}`
      });
    }
    
    // Parse date range
    let startDate, endDate;
    
    if (start_date && end_date) {
      startDate = start_date;
      endDate = end_date;
    } else {
      // Use window
      endDate = new Date().toISOString().split('T')[0];
      const windowDays = {
        '7d': 7,
        '30d': 30,
        '90d': 90,
        '180d': 180,
        '365d': 365,
        'all': 9999
      };
      
      const days = windowDays[window] || 90;
      const start = new Date();
      start.setDate(start.getDate() - days);
      startDate = start.toISOString().split('T')[0];
    }
    
    // Fetch historical data
    const historicalData = await db.getHistoricalChainData(
      chainId,
      startDate,
      endDate
    );
    
    if (!historicalData || historicalData.length === 0) {
      return res.status(404).json({
        error: `No historical data found for chain: ${chainId}`
      });
    }
    
    // Get chain profile
    const profile = chainProfileService.getProfile(chainId);
    
    // Format response
    const response = {
      chain: chainId,
      chain_name: profile.name,
      start_date: startDate,
      end_date: endDate,
      data_points: historicalData.length,
      data: historicalData
    };
    
    res.json(response);
    
  } catch (error) {
    console.error('Error fetching historical data:', error);
    res.status(500).json({
      error: 'Failed to fetch historical data'
    });
  }
});

/**
 * GET /api/chains/:chainId/regime-history
 * Returns history of regime changes
 */
router.get('/:chainId/regime-history', async (req, res) => {
  try {
    const { chainId } = req.params;
    const { limit = 20 } = req.query;
    
    // Validate chain ID
    const validChains = ['ethereum', 'bitcoin', 'arbitrum', 'base'];
    if (!validChains.includes(chainId)) {
      return res.status(400).json({
        error: `Invalid chain ID. Must be one of: ${validChains.join(', ')}`
      });
    }
    
    // Fetch regime shifts
    const regimeShifts = await db.getRegimeShifts(chainId, parseInt(limit));
    
    res.json({
      chain: chainId,
      shifts: regimeShifts
    });
    
  } catch (error) {
    console.error('Error fetching regime history:', error);
    res.status(500).json({
      error: 'Failed to fetch regime history'
    });
  }
});

export default router;
```

### 9.3 Database Access Layer

**File:** `/src/db/index.js`
```javascript
import pg from 'pg';

const { Pool } = pg;

const pool = new Pool({
  host: process.env.DB_HOST || 'localhost',
  port: process.env.DB_PORT || 5432,
  database: process.env.DB_NAME || 'css_db',
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  max: 20,
  idleTimeoutMillis: 30000,
  connectionTimeoutMillis: 2000,
});

class Database {
  
  /**
   * Get latest data for a chain
   */
  async getLatestChainData(chainId) {
    const query = `
      SELECT *
      FROM chain_data
      WHERE chain = $1
      ORDER BY date DESC
      LIMIT 1
    `;
    
    const result = await pool.query(query, [chainId]);
    return result.rows[0] || null;
  }
  
  /**
   * Get historical data for a chain
   */
  async getHistoricalChainData(chainId, startDate, endDate) {
    const query = `
      SELECT *
      FROM chain_data
      WHERE chain = $1
        AND date >= $2
        AND date <= $3
      ORDER BY date ASC
    `;
    
    const result = await pool.query(query, [chainId, startDate, endDate]);
    return result.rows;
  }
  
  /**
   * Get regime shifts for a chain
   */
  async getRegimeShifts(chainId, limit = 20) {
    const query = `
      SELECT 
        shift_date,
        previous_regime,
        current_regime,
        trigger_metric,
        trigger_value
      FROM regime_shifts
      WHERE chain = $1
      ORDER BY shift_date DESC
      LIMIT $2
    `;
    
    const result = await pool.query(query, [chainId, limit]);
    return result.rows;
  }
  
  /**
   * Get most recent regime shift across all chains
   */
  async getMostRecentRegimeShift() {
    const query = `
      SELECT 
        chain,
        shift_date,
        previous_regime,
        current_regime,
        trigger_metric
      FROM regime_shifts
      ORDER BY shift_date DESC
      LIMIT 1
    `;
    
    const result = await pool.query(query);
    return result.rows[0] || null;
  }
  
  /**
   * Get notables for a specific date or date range
   */
  async getNotables(chainId = null, startDate = null, endDate = null) {
    let query = `
      SELECT *
      FROM notables
      WHERE 1=1
    `;
    
    const params = [];
    
    if (chainId) {
      params.push(chainId);
      query += ` AND chain = $${params.length}`;
    }
    
    if (startDate) {
      params.push(startDate);
      query += ` AND date >= $${params.length}`;
    }
    
    if (endDate) {
      params.push(endDate);
      query += ` AND date <= $${params.length}`;
    }
    
    query += ` ORDER BY date DESC, severity DESC`;
    
    const result = await pool.query(query, params);
    return result.rows;
  }
  
  /**
   * Close database connection
   */
  async close() {
    await pool.end();
  }
}

export default new Database();
```

### 11.3 Data Fetching Optimization

**File:** `/src/api/client.js`
```javascript
import axios from 'axios';

const API_BASE_URL = process.env.REACT_APP_API_URL || 'http://localhost:3001/api';

// Create axios instance with defaults
const apiClient = axios.create({
  baseURL: API_BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// Response interceptor for error handling
apiClient.interceptors.response.use(
  (response) => response,
  (error) => {
    console.error('API Error:', error.response?.data || error.message);
    return Promise.reject(error);
  }
);

// Cache layer
const cache = new Map();
const CACHE_DURATION = 5 * 60 * 1000;  // 5 minutes

function getCacheKey(url, params) {
  return `${url}:${JSON.stringify(params || {})}`;
}

function isCacheValid(timestamp) {
  return Date.now() - timestamp < CACHE_DURATION;
}

// Fetch with caching
export async function fetchWithCache(url, params = {}) {
  const cacheKey = getCacheKey(url, params);
  const cached = cache.get(cacheKey);
  
  if (cached && isCacheValid(cached.timestamp)) {
    console.log('Cache hit:', cacheKey);
    return cached.data;
  }
  
  console.log('Cache miss:', cacheKey);
  const response = await apiClient.get(url, { params });
  
  cache.set(cacheKey, {
    data: response.data,
    timestamp: Date.now(),
  });
  
  return response.data;
}

// Clear cache (call when data needs refresh)
export function clearCache() {
  cache.clear();
}

// Specific API functions
export async function fetchChainData(chainId) {
  return fetchWithCache(`/chains/${chainId}/latest`);
}

export async function fetchHistoricalData(chainId, window = '90d') {
  return fetchWithCache(`/chains/${chainId}/historical`, { window });
}

export async function fetchRecentRegimeShift() {
  return fetchWithCache('/landing/recent-regime-shift');
}

export async function fetchNotables(chainId = null) {
  const params = chainId ? { chain: chainId } : {};
  return fetchWithCache('/notables', params);
}

export default apiClient;
```

```markdown
# Quick Start: Read CSS in 30 Seconds

Chain State Snapshots (CSS) shows you **what's happening on blockchains right now**—without price data.

## The Four Signals

Every chain dashboard shows four core signals:

### 1. Regime Status
**What it shows:** Is the network CONGESTED, NORMAL, or COOLING?

**How to read it:**
- 🔴 CONGESTED = High activity relative to recent history
- 🟡 NORMAL = Typical activity levels
- 🟢 COOLING = Lower than recent norms

**What it means:** This is not a buy/sell signal. It's an observation of network conditions relative to that chain's own history.

### 2. Percentile Context
**What it shows:** Where does today rank vs. the past 90 days?

**How to read it:**
- 90th percentile = Higher than 90% of recent days
- 50th percentile = Right in the middle
- 10th percentile = Lower than 90% of recent days

**What it means:** Historical placement, not a prediction of what's coming.

### 3. Moving Averages
**What they show:**
- MA7 (white line) = Last 7 days average
- MA30 (dashed line) = Last 30 days average

**How to read it:**
- Daily above MA7 = Short-term heating
- MA7 above MA30 = Trend is rising
- Lines converging = Activity stabilizing

### 4. Chain-Specific Metrics
Each chain prioritizes different measurements:

**Ethereum:** Gas utilization, failed transactions, fees
**Bitcoin:** Value transferred, settlement size
**Arbitrum/Base:** Transaction count, active users

## What CSS Does NOT Do

❌ Does not predict price movements  
❌ Does not give buy/sell recommendations  
❌ Does not forecast future network behavior  

## What CSS DOES Do

✅ Shows observed network activity  
✅ Provides historical context  
✅ Identifies regime changes  
✅ Explains metrics in plain English
```

### 13.2 Content Loader Service

**File:** `/src/services/contentService.js`
```javascript
import fs from 'fs/promises';
import path from 'path';
import matter from 'gray-matter';
import { marked } from 'marked';

class ContentService {
  constructor() {
    this.contentDir = path.join(process.cwd(), 'content');
    this.cache = new Map();
  }
  
  /**
   * Load and parse markdown content
   */
  async loadContent(contentPath) {
    const cacheKey = contentPath;
    
    // Check cache
    if (this.cache.has(cacheKey)) {
      return this.cache.get(cacheKey);
    }
    
    try {
      // Read file
      const fullPath = path.join(this.contentDir, `${contentPath}.md`);
      const fileContent = await fs.readFile(fullPath, 'utf-8');
      
      // Parse frontmatter
      const { data: frontmatter, content } = matter(fileContent);
      
      // Convert markdown to HTML
      const html = marked(content);
      
      // Validate no forbidden language
      this.validateContent(content);
      
      const result = {
        frontmatter,
        content,
        html
      };
      
      // Cache result
      this.cache.set(cacheKey, result);
      
      return result;
      
    } catch (error) {
      console.error(`Error loading content: ${contentPath}`, error);
      throw new Error(`Content not found: ${contentPath}`);
    }
  }
  
  /**
   * Validate content for legal compliance
   */
  validateContent(content) {
    const forbiddenTerms = [
      'will',
      'should',
      'expect to',
      'likely to',
      'predicted',
      'forecasted',
      'you should',
      'we recommend',
      'buy now',
      'sell now'
    ];
    
    const lowerContent = content.toLowerCase();
    
    for (const term of forbiddenTerms) {
      if (lowerContent.includes(term)) {
        throw new Error(
          `Content validation failed: Forbidden term "${term}" found in content`
        );
      }
    }
  }
  
  /**
   * List all content in a directory
   */
  async listContent(directory) {
    try {
      const dirPath = path.join(this.contentDir, directory);
      const files = await fs.readdir(dirPath);
      
      const contentList = [];
      
      for (const file of files) {
        if (file.endsWith('.md')) {
          const slug = file.replace('.md', '');
          const content = await this.loadContent(`${directory}/${slug}`);
          
          contentList.push({
            slug,
            ...content.frontmatter
          });
        }
      }
      
      // Sort by order if specified
      contentList.sort((a, b) => {
        if (a.order !== undefined && b.order !== undefined) {
          return a.order - b.order;
        }
        return 0;
      });
      
      return contentList;
      
    } catch (error) {
      console.error(`Error listing content: ${directory}`, error);
      return [];
    }
  }
  
  /**
   * Clear cache
   */
  clearCache() {
    this.cache.clear();
  }
}

export default new ContentService();
```
